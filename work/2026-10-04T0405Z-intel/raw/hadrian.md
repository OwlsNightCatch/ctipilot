  const now = Date.now();

  // Selectors
  const popup = document.querySelector('.pop-up-banner-outer');
  const popupWrapper = document.querySelector('.pop-up-banner-wrapper');
  const exitIntent = document.querySelector('.intent-banner-outer');

  const KEYS = {
    popup: 'popupBannerDismissed',
    intent: 'intentBannerDismissed'
  };

  // Utility functions
  const isDismissed = (key) => {
    const ts = localStorage.getItem(key);
    return ts && now - parseInt(ts) < THIRTY_DAYS;
  };

  const dismissBanner = (banner, key) => {
    banner.classList.remove('cc-active');
    localStorage.setItem(key, now.toString());
  };

  const hasContent = (wrapper) => {
    if (!wrapper || wrapper.children.length === 0) return false;
    return !wrapper.querySelector('.w-dyn-empty');
  };

  const isPopupVisible = () => popup?.classList.contains('cc-active');
  const isExitVisible = () => exitIntent?.classList.contains('cc-active');

  let popupAlreadyShown = false;
  let exitTriggered = false;

  const tryShowPopup = () => {
    if (
      popupAlreadyShown ||
      isDismissed(KEYS.popup) ||
      !hasContent(popupWrapper) ||
      isExitVisible()
    ) return;

    setTimeout(() => {
      if (
        !popupAlreadyShown &&
        !isDismissed(KEYS.popup) &&
        hasContent(popupWrapper) &&
        !isExitVisible()
      ) {
        popup.classList.add('cc-active');
        popupAlreadyShown = true;
      }
    }, 5000);
  };

  if (!popup?.querySelector('.w-dyn-empty')) {
    tryShowPopup();
  }

  const showExitIntent = () => {
    if (
      exitTriggered ||
      !hasContent(intentWrapper) ||
      isDismissed(KEYS.intent) ||
      isPopupVisible()
    ) return;

    exitTriggered = true;

    setTimeout(() => {
      if (!isPopupVisible()) {
        exitIntent?.classList.add('cc-active');
      }
    }, 2000);
  };

  if (!exitIntent?.querySelector('.w-dyn-empty')) {
    if (window.innerWidth > 768) {
      document.addEventListener("mouseout", (e) => {
        const leavingTop = !e.relatedTarget && e.clientY <= 0;
        const leavingLeft = !e.relatedTarget && e.clientX <= 0;

        if ((leavingTop || leavingLeft) && !exitTriggered) {
          showExitIntent();
        }
      });

      }
    });

    else if (direction === "next" && nextIndex === 3) {
      currentImage.style.transform = getTranslationValue(3);
      nextImage.style.transform = getTranslationValue(3);
      currentContent.style.transform = getTranslationValue(3);
      currentContent.style.opacity = "0";
      nextContent.style.transform = getTranslationValue(3);
      nextContent.style.opacity = "1";
    }
    // Previous slides
    else if (direction === "prev" && nextIndex === 2) {
      currentImage.style.transform = getTranslationValue(2);
      nextImage.style.transform = getTranslationValue(2);
      currentContent.style.transform = getTranslationValue(2);
      currentContent.style.opacity = "0";
      nextContent.style.transform = getTranslationValue(2);
      nextContent.style.opacity = "1";
    } else if (direction === "prev" && nextIndex === 1) {
      currentImage.style.transform = getTranslationValue(1);
      nextImage.style.transform = getTranslationValue(1);
      currentContent.style.transform = getTranslationValue(1);
      currentContent.style.opacity = "0";
      nextContent.style.transform = getTranslationValue(1);
      nextContent.style.opacity = "1";
    } else if (direction === "prev" && nextIndex === 0) {
      currentImage.style.transform = getTranslationValue(0);
      nextImage.style.transform = getTranslationValue(0);
      currentContent.style.transform = getTranslationValue(0);
      currentContent.style.opacity = "0";
      nextContent.style.transform = getTranslationValue(0);
      nextContent.style.opacity = "1";
    }

    // Update pagination numbers with animation
    updatePaginationNumbers(nextIndex, direction);

    // Update current slide index
    currentSlideIndex = nextIndex;

    // Update arrow states
    updateArrowStates();
  }

  // Add click event listeners to arrows
  if (bottomArrow) {
    bottomArrow.addEventListener("click", () => changeSlide("next"));
  }

  if (topArrow) {
    topArrow.addEventListener("click", () => changeSlide("prev"));
  }

  // Initialize arrow states
  updateArrowStates();
});
</script>
